interface Avatar3DProps {
  build: BodyBuild;
  muscles: Partial<Record<Muscle, number>>;
  outfit: Outfit;
  paused?: boolean;
  animate?: boolean;
  studio?: boolean;
  className?: string;
  look?: AvatarLook;
  focus?: AvatarFocus;
  yaw?: number;
  onYawChange?: (angle: number) => void;
}

type FigureParams = Pick<Avatar3DProps, 'build' | 'muscles' | 'outfit' | 'look'>;

interface ImportedFigure {
  root: THREE.Group;
  surfaces: THREE.SkinnedMesh[];
  nailSurfaces: THREE.SkinnedMesh[];
  lipSurfaces: THREE.SkinnedMesh[];
  eyes: THREE.SkinnedMesh | undefined;
  face?: AvatarFaceForm;
  regionalDermal: boolean;
  garmentFit: number;
  sleeveRepair: number;
  lashes?: THREE.SkinnedMesh;
  garments: THREE.SkinnedMesh[];
  wardrobe?: AthleteWardrobe;
  materials: AvatarMaterials;
  hair: THREE.MeshStandardMaterial;
  buzzHair: THREE.MeshStandardMaterial;
  hairStyles: Record<HairStyle, THREE.Object3D>;
  groomAnchor: THREE.Group;
  strandGroom?: { crop: THREE.Mesh; swept: THREE.Mesh; material: THREE.MeshPhysicalMaterial; layered: boolean; directional?: boolean };
  sculpt?: ReturnType<typeof createAvatarPhysique>;
  seams?: AvatarGarmentSeams;
  idle?: AvatarIdleRig;
  idleBreath: number;
  realistic: boolean;
  groomed: boolean;
  visor: THREE.Mesh;
  visorMaterial: THREE.MeshStandardMaterial;
  visorEdge: THREE.Mesh;
  visorEdgeMaterial: THREE.MeshStandardMaterial;
}

/**
 * Local rigged athlete, with a procedural human fallback.
 * No external image CDN or access to the camera stream.
 */
export function Avatar3D({ build, muscles, outfit, look = DEFAULT_AVATAR_LOOK, focus = 'full',
  yaw = INITIAL_AVATAR_YAW, onYawChange, paused = false, animate = false, studio: studioView = false, className = '' }: Avatar3DProps) {
  const canvas = useRef<HTMLCanvasElement>(null);
  const runtime = useRef<{
    renderer: THREE.WebGLRenderer;
    scene: THREE.Scene;
    camera: THREE.PerspectiveCamera;
    figureRoot: THREE.Group;
    current: THREE.Group | null;
    stopped: () => void;
    redraw: () => void;
    syncMotion: () => void;
    setView: (next: AvatarFocus) => void;
    setYaw: (next: number) => void;
    setAccent: (color: string) => void;
  } | null>(null);
  const pausedRef = useRef(paused);
  const animateRef = useRef(animate);
  const studioRef = useRef(studioView);
  const onYawChangeRef = useRef(onYawChange);
  onYawChangeRef.current = onYawChange;
  const figureSettingsRef = useRef<FigureParams>({ build, muscles, outfit, look });
  const importedRef = useRef<ImportedFigure | null>(null);
  figureSettingsRef.current = { build, muscles, outfit, look };
  const [available, setAvailable] = useState(true);
  const [modelSource, setModelSource] = useState<'procedural' | 'cc0'>('procedural');
  const [modelAsset, setModelAsset] = useState<'procedural' | 'legacy' | 'athlete'>('procedural');
  const [skinMapReady, setSkinMapReady] = useState(false);
  const [realistic, setRealistic] = useState(false);
  const [groomed, setGroomed] = useState(false);
  const [hairMaps, setHairMaps] = useState(0);
  const [detailMaps, setDetailMaps] = useState(0);

  useEffect(() => {
    pausedRef.current = paused;
    animateRef.current = animate;
    studioRef.current = studioView;
    runtime.current?.syncMotion();
    runtime.current?.redraw();
  }, [paused, animate, studioView]);

  useEffect(() => {
    const element = canvas.current as HTMLCanvasElement;
    if (!element) return;
    // Fail before allocating WebGL or attaching listeners. A partial setup
    // used to leave visibility callbacks referring to an uninitialized loop.
    if (typeof IntersectionObserver !== 'function' || typeof ResizeObserver !== 'function') {
      setAvailable(false);
      return;
    }
    const telemetryId = runtimeDiagnostics.attachAvatar();
    let renderer: THREE.WebGLRenderer | null = null;
    let studio: THREE.WebGLRenderTarget | null = null;
    let portraitShadow: THREE.DirectionalLightShadow | undefined;
    let intersectsViewport = true;
    let visible = !document.hidden;
    let disposed = false;
    let rotation = clampAvatarYaw(yaw);
    let pointerStartYaw = rotation;
    let activePointer: number | null = null;
    let pointerX = 0;
    let isDragging = false;
    // Only the studio opts into a bounded idle motion loop.

    try {
      renderer = new THREE.WebGLRenderer({
        canvas: element,
        alpha: true,
        antialias: true,
        powerPreference: 'low-power',
        preserveDrawingBuffer: false,
      });
      renderer.setPixelRatio(avatarPixelRatio(document.documentElement.dataset.bfGraphics, window.devicePixelRatio));
      renderer.outputColorSpace = THREE.SRGBColorSpace;
      const lighting = avatarLookDev(import.meta.env.DEV ? new URLSearchParams(window.location.search).get('avatarLookDev') : null);
      renderer.toneMapping = lighting.toneMapping;
      renderer.toneMappingExposure = lighting.exposure;
      element.dataset.lookDev = lighting.id;
      renderer.shadowMap.type = THREE.PCFShadowMap;
      const scene = new THREE.Scene();
      try {
        studio = createAvatarStudio(renderer);
        scene.environment = studio.texture;
        scene.environmentIntensity = lighting.fallbackIbl;
        element.dataset.studioIbl = 'synthetic';
      } catch (error) {
        console.warn('BODYFORGE studio IBL unavailable; keeping direct lights:', error);
      }
      const camera = new THREE.PerspectiveCamera(35, 1, 0.1, 60);
      camera.position.set(0, 2.40, 8.35);
      camera.lookAt(0, 2.38, 0);
      const hemisphere = new THREE.HemisphereLight(lighting.sky, lighting.ground, lighting.hemisphere);
      scene.add(hemisphere);
      const warm = new THREE.DirectionalLight(lighting.keyColor, lighting.key);
      scene.add(warm, warm.target);
      portraitShadow = warm.shadow;
      configureAvatarKey(warm, focus);
      const fill = new THREE.DirectionalLight(lighting.fillColor, lighting.fill);
      fill.position.set(4.2, 5.0, 5);
      fill.target.position.set(0, 3, 0);
      scene.add(fill, fill.target);
      const rim = new THREE.DirectionalLight(lighting.rimColor, lighting.rim);
      rim.position.set(3, 6, -4.5);
      rim.target.position.set(0, 3, 0);
      scene.add(rim, rim.target);

      const stage = new THREE.Group();
      scene.add(stage);
      const pedestalMaterial = new THREE.MeshStandardMaterial({ color: 0x213245, metalness: .7, roughness: .4 });
      const edgeMaterial = new THREE.MeshStandardMaterial({ color: 0xf1bc65, metalness: .65, roughness: .32, emissive: 0x705025, emissiveIntensity: .24 });
      const base = new THREE.Mesh(new THREE.CylinderGeometry(1.65, 1.78, 0.24, 44), pedestalMaterial);
      base.position.y = -0.12;
      base.receiveShadow = true;
      stage.add(base);
      const edge = new THREE.Mesh(new THREE.TorusGeometry(1.66, .045, 8, 64), edgeMaterial);
      edge.rotation.x = Math.PI / 2;
      edge.position.y = .015;
      stage.add(edge);
      const innerGlow = new THREE.Mesh(
        new THREE.CylinderGeometry(1.43, 1.43, .025, 40),
        new THREE.MeshBasicMaterial({ color: 0xf8c978, transparent: true, opacity: .045 }),
      );
      innerGlow.position.y = .015;
      stage.add(innerGlow);

      const figureRoot = new THREE.Group();
      figureRoot.rotation.y = rotation;
      stage.add(figureRoot);
      let currentView = focus;
      const setView = (next: AvatarFocus) => {
        currentView = next;
        configureAvatarKey(warm, next);
        if (next === 'portrait') {
          camera.position.set(0, 4.37, 2.55);
          camera.lookAt(0, 4.42, 0);
        } else if (next === 'torso') {
          camera.position.set(0, 3.27, 4.6);
          camera.lookAt(0, 3.23, 0);
        } else if (next === 'legs') {
          configureAvatarLegCamera(camera);
        } else {
          camera.position.set(0, 2.40, 8.35);
          camera.lookAt(0, 2.38, 0);
        }
        element.dataset.avatarView = next;
        syncMotion();
        redraw();
      };
      const setYaw = (next: number) => {
        rotation = clampAvatarYaw(next);
        element.dataset.avatarYaw = String(rotation);
        figureRoot.rotation.y = rotation;
        redraw();
      };
      const setAccent = (color: string) => {
        const safe = /^#[0-9a-f]{6}$/i.test(color) ? color : '#55e3f2';
        edgeMaterial.color.set(safe);
        if (lighting.accentRim) rim.color.copy(avatarRimColor(safe));
        redraw();
      };

      function redraw() {
        // ResizeObserver, GLB loading and profile updates can redraw without RAF.
        // Apply the same visibility/camera pause boundary to these entry points.
        if (!disposed && renderer && visible && !document.hidden && !pausedRef.current) {
          const rect = element.getBoundingClientRect();
          const w = Math.max(1, Math.round(rect.width));
          const h = Math.max(1, Math.round(rect.height));
          if (w > 1 && h > 1) {
            const ratio = avatarPixelRatio(document.documentElement.dataset.bfGraphics, window.devicePixelRatio);
            if (renderer.getPixelRatio() !== ratio) renderer.setPixelRatio(ratio);
            element.dataset.renderScale = String(ratio);
            if (element.width !== Math.round(w * renderer.getPixelRatio()) ||
                element.height !== Math.round(h * renderer.getPixelRatio())) {
              renderer.setSize(w, h, false);
              camera.aspect = w / h;
              camera.updateProjectionMatrix();
            }
            // One fitted key-light depth map, only when a visible frame is
            // requested. Economy and non-studio full-body views stay cheap.
            const shadows = avatarShadowsEnabled(document.documentElement.dataset.bfGraphics, currentView, studioRef.current);
            renderer.shadowMap.enabled = warm.castShadow = shadows;
            element.dataset.studioShadows = shadows ? currentView : 'off';
            const wardrobe = importedRef.current?.wardrobe;
            element.dataset.shirtState = wardrobe ? (wardrobe.shirtVisible ? 'on' : 'off') : 'fallback';
            renderer.render(scene, camera);
            runtimeDiagnostics.recordAvatarFrame(telemetryId, performance.now(), {
              drawCalls: renderer.info.render.calls,
              geometries: renderer.info.memory.geometries,
              textures: renderer.info.memory.textures,
            });
          }
        }
      }

      const onPointerDown = (event: PointerEvent) => {
        activePointer = event.pointerId;
        pointerStartYaw = rotation;
        pointerX = event.clientX;
        isDragging = true;
        element.setPointerCapture(event.pointerId);
      };
      const onPointerMove = (event: PointerEvent) => {
        if (event.pointerId !== activePointer || !isDragging) return;
        rotation = stepAvatarYaw(rotation, (event.clientX - pointerX) * .011);
        pointerX = event.clientX;
        element.dataset.avatarYaw = String(rotation);
        figureRoot.rotation.y = rotation;
        redraw();
      };
      const onPointerUp = (event: PointerEvent) => {
        if (event.pointerId === activePointer) {
          const changed = isDragging && rotation !== pointerStartYaw;
          isDragging = false;
          activePointer = null;
          if (changed) onYawChangeRef.current?.(rotation);
          if (element.hasPointerCapture(event.pointerId)) element.releasePointerCapture(event.pointerId);
        }
      };
      const onArrowKey = (event: KeyboardEvent) => {
        if (event.key !== 'ArrowLeft' && event.key !== 'ArrowRight') return;
        event.preventDefault();
        rotation = stepAvatarYaw(rotation, event.key === 'ArrowLeft' ? -.12 : .12);
        element.dataset.avatarYaw = String(rotation);
        figureRoot.rotation.y = rotation;
        onYawChangeRef.current?.(rotation);
        redraw();
      };
      const onContextLost = (event: Event) => {
        event.preventDefault();
        console.warn('BODYFORGE 3D context lost');
        // A fallback replaces the canvas without unmounting this component.
        // Stop its runtime now, before later profile updates can redraw it.
        currentRuntime.stopped();
        setAvailable(false);
      };
      element.addEventListener('pointerdown', onPointerDown);
      element.addEventListener('pointermove', onPointerMove);
      element.addEventListener('pointerup', onPointerUp);
      element.addEventListener('pointercancel', onPointerUp);
      element.addEventListener('keydown', onArrowKey);
      element.addEventListener('webglcontextlost', onContextLost);

      const onVisibility = () => {
        visible = !document.hidden && intersectsViewport && element.getBoundingClientRect().height > 0;
        syncMotion();
        if (visible) redraw();
      };
      document.addEventListener('visibilitychange', onVisibility);
      window.addEventListener(GRAPHICS_CHANGE_EVENT, onVisibility);
      window.addEventListener('resize', onVisibility);
      const observer = new IntersectionObserver((entries) => {
        intersectsViewport = entries[0]?.isIntersecting === true;
        onVisibility();
      }, { threshold: .01 });
      observer.observe(element);
      // CSS visibility/size changes need a fresh draw when the stage returns.
      const resize = new ResizeObserver(onVisibility);
      resize.observe(element);

      const motionQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
      const idleClock = new AvatarIdleClock();
      let moving = false;
      let motionFailed = false;
      const frameLoop = new ActiveFrameLoop(time => {
        const seconds = idleClock.tick(time);
        const imported = importedRef.current;
        if (!moving || !imported?.idle || disposed || document.hidden) return;
        if (seconds === undefined) return;
        try {
          const pose = sampleAvatarIdle(seconds);
          imported.idle.pose(pose);
          imported.idleBreath = pose.breath;
          applyAvatarGrowth(imported.root, figureSettingsRef.current.muscles, pose.breath);
          imported.seams?.update(imported.root);
          element.dataset.idleBlink = pose.blink.toFixed(3);
          element.dataset.idleBreath = pose.breath.toFixed(3);
          redraw();
        } catch (error) {
          motionFailed = true;
          syncMotion();
          console.warn('BODYFORGE idle pose unavailable; retaining static avatar:', error);
        }
      }, callback => window.requestAnimationFrame(callback), id => window.cancelAnimationFrame(id));
      const onMotionPreference = () => {
        const wasMoving = moving;
        syncMotion();
        if (wasMoving !== moving) redraw();
      };
      motionQuery.addEventListener('change', onMotionPreference);

      function syncMotion() {
        const shown = visible && !document.hidden && !pausedRef.current && !disposed;
        const imported = importedRef.current;
        const enabled = shown && animateRef.current && Boolean(imported?.idle) && !motionFailed &&
          !motionQuery.matches && document.documentElement.dataset.bfGraphics !== 'economy' &&
          currentView !== 'legs';
        if (moving !== enabled) {
          moving = enabled;
          idleClock.reset();
          frameLoop.setActive(enabled);
          if (!enabled && imported?.idle) {
            imported.idle.reset();
            imported.idleBreath = 0;
            applyAvatarGrowth(imported.root, figureSettingsRef.current.muscles);
            imported.seams?.update(imported.root);
            element.dataset.idleBlink = '0';
            element.dataset.idleBreath = '0';
          }
        }
        runtimeDiagnostics.setAvatarAnimating(telemetryId, enabled);
        element.dataset.renderLoop = shown ? enabled ? 'idle-motion' : 'on-demand' : 'paused';
      }
      const currentRuntime = {
        renderer,
        setView,
        setYaw,
        setAccent,
        scene,
        camera,
        figureRoot,
        current: null as THREE.Group | null,
        redraw,
        syncMotion,
        stopped: () => {
          if (disposed) return;
          disposed = true;
          frameLoop.stop();
          motionQuery.removeEventListener('change', onMotionPreference);
          element.dataset.renderLoop = 'stopped';
          if (runtime.current === currentRuntime) {
            runtime.current = null;
            importedRef.current = null;
          }
          element.removeEventListener('pointerdown', onPointerDown);
          element.removeEventListener('pointermove', onPointerMove);
          element.removeEventListener('pointerup', onPointerUp);
          element.removeEventListener('pointercancel', onPointerUp);
          element.removeEventListener('keydown', onArrowKey);
          element.removeEventListener('webglcontextlost', onContextLost);
          document.removeEventListener('visibilitychange', onVisibility);
          window.removeEventListener(GRAPHICS_CHANGE_EVENT, onVisibility);
          window.removeEventListener('resize', onVisibility);
          observer.disconnect();
          resize.disconnect();
          disposeThreeTree(scene);
          portraitShadow?.dispose();
          scene.environment = null;
          studio?.dispose();
          studio = null;
          runtimeDiagnostics.detachAvatar(telemetryId);
          // Keep the canvas context valid across React StrictMode's
          // development-only effect cleanup/remount cycle.
          renderer?.dispose();
        },
      };
      runtime.current = currentRuntime;
      setView(focus);
      setYaw(yaw);
      setAccent(lookColors(look).trim);
      onVisibility();

      // Licensed CC0 MakeHuman asset. Swapped in only when the complete scene,
      // clothing and materials are ready; procedural human remains the fallback.
      const glbStartedAt = performance.now();
      const loadStrandGroom = (imported: ImportedFigure, onFallback: () => void, layered = true, allowReview = true) => {
        const isCurrent = () => !disposed && currentRuntime.current === imported.root;
        const recover = () => {
          if (!isCurrent()) return;
          if (layered) loadStrandGroom(imported, onFallback, false);
          else onFallback();
        };
        if (import.meta.env.DEV && layered && allowReview &&
            (window.location.hostname === '127.0.0.1' || window.location.hostname === 'localhost') &&
            new URLSearchParams(window.location.search).get('avatarGroomV3') === '1') {
          void import('../lib/avatarDirectionalGroomReview')
            .then(({ loadDirectionalGroomReview }) => loadDirectionalGroomReview())
            .then(({ groom, crop, swept, material }) => {
              if (!isCurrent()) {
                disposeThreeTree(groom);
                return;
              }
              imported.groomAnchor.add(groom);
              imported.strandGroom = { crop, swept, material, layered: true, directional: true };
              styleAvatarHair(imported, figureSettingsRef.current.look ?? DEFAULT_AVATAR_LOOK);
              element.dataset.groomGeometry = 'strands';
              element.dataset.groomRevision = 'directional-v3';
              redraw();
            }, () => {
              if (isCurrent()) loadStrandGroom(imported, onFallback, true, false);
            });
          return;
        }
        const url = layered ? '/assets/bodyforge-strand-groom-v2.glb' : '/assets/bodyforge-strand-groom.glb';
        new GLTFLoader().load(url, ({ scene: groom }) => {
          if (disposed || currentRuntime.current !== imported.root) {
            disposeThreeTree(groom);
            return;
          }
          const crop = groom.getObjectByName('BODYFORGE_Groom_crop');
          const swept = groom.getObjectByName('BODYFORGE_Groom_swept');
          const valid = (mesh: THREE.Object3D | undefined): mesh is THREE.Mesh =>
            mesh instanceof THREE.Mesh && mesh.geometry.hasAttribute('color') && mesh.geometry.hasAttribute('normal');
          if (!valid(crop) || !valid(swept) || (layered &&
              (groom.children.length !== 2 || !isLayeredAvatarGroom(crop) || !isLayeredAvatarGroom(swept)))) {
            disposeThreeTree(groom);
            recover();
            return;
          }
          const material = layered ? createLayeredGroomMaterial() :
            new THREE.MeshPhysicalMaterial({ name: 'BODYFORGE strand hair',
              vertexColors: true, side: THREE.DoubleSide, roughness: .48, specularIntensity: .45 });
          const oldMaterials = new Set<THREE.Material>();
          for (const mesh of [crop, swept]) {
            for (const old of Array.isArray(mesh.material) ? mesh.material : [mesh.material]) oldMaterials.add(old);
            mesh.material = material;
            mesh.castShadow = true;
          }
          oldMaterials.forEach(old => old.dispose());
          imported.groomAnchor.add(groom);
          imported.strandGroom = { crop, swept, material, layered };
          styleAvatarHair(imported, figureSettingsRef.current.look ?? DEFAULT_AVATAR_LOOK);
          element.dataset.groomGeometry = 'strands';
          element.dataset.groomRevision = layered ? 'layered-v2' : 'v1';
          redraw();
        }, undefined, recover);
      };
      const loadFigure = ([url, ...fallback]: readonly string[]) => new GLTFLoader().load(
        url,
        async (gltf) => {
          if (disposed || runtime.current !== currentRuntime) {
            disposeThreeTree(gltf.scene);
            return;
          }
          try {
            const preferred = url.endsWith('/bodyforge-athlete-efficient.glb');
            let scanReview: ((source: THREE.Object3D) => number) | undefined;
            let scanColorUrl: string | undefined;
            if (import.meta.env.DEV && preferred &&
                (window.location.hostname === '127.0.0.1' || window.location.hostname === 'localhost') &&
                new URLSearchParams(window.location.search).get('avatarHeadVolume') === '1') {
              try {
                const { parseHeadVolumeRecipe, applyAvatarHeadVolume } = await import('../lib/avatarHeadVolumeReview');
                const response = await fetch('/_bodyforge/review/head-volume-v4.json', {
                  cache: 'no-store', credentials: 'same-origin', signal: AbortSignal.timeout(6000),
                });
                const recipe = response.ok ? parseHeadVolumeRecipe(await response.json()) : undefined;
                if (recipe) scanReview = source => applyAvatarHeadVolume(source, recipe);
              } catch {
                // Optional local recipe unavailable: retain the verified base head.
              }
              if (disposed || runtime.current !== currentRuntime) {
                disposeThreeTree(gltf.scene);
                return;
              }
            } else if (import.meta.env.DEV && preferred && new URLSearchParams(window.location.search).get('avatarScanReview') === '1') {
              try {
                scanReview = (await import('../lib/avatarScanTransfer')).applyAvatarScanTransfer;
                if (new URLSearchParams(window.location.search).get('avatarScanColor') === '1') {
                  scanColorUrl = (await import('../lib/avatarScanColor')).scanColorAtlasUrl;
                }
              } catch {
                // A missing optional development asset must leave the base face available.
              }
              if (disposed || runtime.current !== currentRuntime) {
                disposeThreeTree(gltf.scene);
                return;
              }
            }
            const imported = buildImportedFigure(gltf.scene, figureSettingsRef.current, preferred, scanReview);
            if (currentRuntime.current) {
              currentRuntime.figureRoot.remove(currentRuntime.current);
              disposeThreeTree(currentRuntime.current);
            }
            currentRuntime.figureRoot.add(imported.root);
            currentRuntime.current = imported.root;
            importedRef.current = imported;
            setModelSource('cc0');
            setModelAsset(imported.garments.length > 0 ? 'athlete' : 'legacy');
            setRealistic(imported.realistic);
            setGroomed(imported.groomed);
            setSkinMapReady(false);
            setDetailMaps(0);
            setHairMaps(0);
            delete element.dataset.groomGeometry;
            delete element.dataset.groomRevision;
            delete element.dataset.scanTorsoRelief;
            delete element.dataset.athleticReview;
            delete element.dataset.muscleReliefReview;
            delete element.dataset.ocularDetail;
            delete element.dataset.scanDetail;
            delete element.dataset.normalSurface;
            delete element.dataset.skinSurface;
            element.dataset.dermalRegions = imported.regionalDermal ? 'regional-v3' : 'base';
            if (import.meta.env.DEV && imported.materials.body.userData.BODYFORGE_diffusionReview === true) {
              element.dataset.skinDiffusion = 'angular-lut-v1';
            } else delete element.dataset.skinDiffusion;
            element.dataset.bodyDermal = String(imported.face?.body.userData.bodyDermalRevision ?? 'base');
            delete element.dataset.lipResponse;
            delete element.dataset.metricSkinReview;
            element.dataset.fabricMaps = '0';
            delete element.dataset.trainerDetail;
            element.dataset.garmentSeams = String(imported.seams?.loopCount ?? 0);
            element.dataset.garmentFit = String(imported.garmentFit);
            if (typeof imported.root.userData.BODYFORGE_scanTorsoRelief === 'string') {
              element.dataset.scanTorsoRelief = imported.root.userData.BODYFORGE_scanTorsoRelief;
            }
            if (typeof imported.root.userData.BODYFORGE_athleticReview === 'string') {
              element.dataset.athleticReview = imported.root.userData.BODYFORGE_athleticReview;
            }
            if (typeof imported.root.userData.BODYFORGE_muscleReliefReview === 'string') {
              element.dataset.muscleReliefReview = imported.root.userData.BODYFORGE_muscleReliefReview;
            }
            element.dataset.sleeveRepair = String(imported.sleeveRepair);
            element.dataset.garmentPatches = String(imported.seams?.patchTriangles ?? 0);
            element.dataset.avatarFace = imported.face ? 'sculpted' : 'base';
            element.dataset.faceContour = imported.face?.body.userData.facialContour === 'lower-face-contour-v3' ? 'lower-face-contour-v3' : 'base';
            if (import.meta.env.DEV && imported.face?.body.userData.scanTransfer) {
              element.dataset.scanTransfer = String(imported.face.body.userData.scanTransfer);
            } else delete element.dataset.scanTransfer;
            if (import.meta.env.DEV && imported.face?.body.userData.headVolumeReview) {
              element.dataset.headVolumeReview = String(imported.face.body.userData.headVolumeReview);
            } else delete element.dataset.headVolumeReview;
            delete element.dataset.browDetail;
            delete element.dataset.browRevision;
            element.dataset.lashDetail = imported.lashes ? 'skinned-fibres' : 'base';
            element.dataset.idleRig = imported.idle ? 'fitted' : 'static';
            syncMotion();
            if (preferred) {
              new HDRLoader().load('/assets/athlete-studio.hdr', texture => {
                try {
                  if (disposed || !renderer || currentRuntime.current !== imported.root) return;
                  const measured = createMeasuredAvatarStudio(renderer, texture);
                  const previous = studio;
                  studio = measured;
                  scene.environment = measured.texture;
                  scene.environmentIntensity = lighting.measuredIbl;
                  scene.environmentRotation.y = lighting.iblRotation;
                  previous?.dispose();
                  element.dataset.studioIbl = 'measured-hdr';
                  redraw();
                } catch (error) {
                  console.warn('BODYFORGE measured studio unavailable, keeping fallback:', error);
                } finally {
                  texture.dispose();
                }
              }, undefined, () => { /* The local synthetic studio remains usable. */ });
            }
            const loadCardMaps = () => {
              if (disposed || currentRuntime.current !== imported.root) return;
              element.dataset.groomGeometry = 'cards';
              for (const [kind, file] of [['albedo', 'athlete-hair.png'], ['normal', 'athlete-hair-normal.png']] as const) {
                new THREE.TextureLoader().load('/assets/' + file, (texture) => {
                  if (disposed || currentRuntime.current !== imported.root) {
                    texture.dispose();
                    return;
                  }
                  applyAvatarHairMap(imported.hair, texture, kind);
                  setHairMaps((count) => count + 1);
                  redraw();
                }, undefined, () => { /* Keep the opaque haircut if an optional map fails. */ });
              }
            };
            if (imported.groomed) {
              if (preferred) loadStrandGroom(imported, loadCardMaps);
              else loadCardMaps();
            }
            if (imported.garments.length > 0) {
              const loadSkin = ([file, ...fallback]: readonly string[]) => {
                if (!file || disposed || currentRuntime.current !== imported.root) return;
                new THREE.TextureLoader().load(file, (texture) => {
                if (disposed || currentRuntime.current !== imported.root) {
                  texture.dispose();
                  return;
                }
                if (!acceptsSkinAtlas(file, texture.image)) {
                  texture.dispose();
                  loadSkin(fallback);
                  return;
                }
                // glTF UVs use the opposite vertical origin to TextureLoader.
                texture.flipY = false;
                texture.colorSpace = THREE.SRGBColorSpace;
                texture.anisotropy = Math.min(4, renderer?.capabilities.getMaxAnisotropy() ?? 1);
                imported.materials.body.map = texture;
                if (imported.realistic) {
                  imported.materials.lips.map = texture;
                  imported.materials.lips.roughness = imported.materials.lips.roughnessMap ? 1 : imported.regionalDermal ? .58 : .40;
                  imported.materials.lips.needsUpdate = true;
                }
                imported.materials.body.needsUpdate = true;
                styleImportedFigure(imported, figureSettingsRef.current);
                setSkinMapReady(true);
                element.dataset.skinSurface = file.split('/').at(-1);
                redraw();
                }, undefined, () => loadSkin(fallback));
              };
              const skinAtlases = avatarSkinAtlases(preferred && imported.regionalDermal, imported.realistic);
              // Local derivative only; no raw scan or private folder exposed.
              // The optional file is git-ignored and not part of production.
              const localTorsoReview = import.meta.env.DEV && preferred &&
                (window.location.hostname === '127.0.0.1' || window.location.hostname === 'localhost') &&
                imported.root.userData.BODYFORGE_scanTorsoRelief === 'BF_ARTEC_TORSO_RELIEF_DEV_V1';
              const torsoPhotoReview = localTorsoReview &&
                new URLSearchParams(window.location.search).get('avatarScanTorsoColor') === '1';
              const torsoNormalReview = localTorsoReview &&
                new URLSearchParams(window.location.search).get('avatarScanTorsoNormal') === '1';
              loadSkin(torsoPhotoReview ? ['/src/lib/athlete-torso-color-review.png', ...skinAtlases]
                : import.meta.env.DEV && scanColorUrl && imported.face?.body.userData.scanTransfer === 'scan-depth-review-v1'
                ? [scanColorUrl, ...skinAtlases] : skinAtlases);
              if (imported.realistic) {
                const isCurrent = () => !disposed && currentRuntime.current === imported.root;
                const loadDetailMap = (kind: AvatarDetailMap, file: string) => {
                  if (!isCurrent()) return;
                  const scanNormal = import.meta.env.DEV && kind === 'normal' && file === 'athlete-torso-normal-review.png';
                  const bakedNormal = kind === 'normal' && file === 'athlete-face-relief-normal.png';
                  const recover = () => {
                    if (!isCurrent()) return;
                    if (scanNormal) loadDetailMap('normal', 'athlete-face-relief-normal.png');
                    else if (bakedNormal) loadDetailMap('normal', 'athlete-normal.png');
                  };
                  new THREE.TextureLoader().load((scanNormal ? '/src/lib/' : '/assets/') + file, (texture) => {
                    if (!isCurrent()) {
                      texture.dispose();
                      return;
                    }
                    const image = texture.image as { width?: number; height?: number } | undefined;
                    if ((bakedNormal || scanNormal) && (image?.width !== 1024 || image.height !== 1024)) {
                      texture.dispose();
                      recover();
                      return;
                    }
                    applyAvatarDetailMap(imported.materials, kind, texture);
                    if (kind === 'normal') element.dataset.normalSurface = scanNormal ? 'scan-torso-bake-v1' : bakedNormal ? 'sculpt-bake-v1' : 'base';
                    if (kind === 'eyes') element.dataset.ocularDetail = 'atlas';
                    setDetailMaps(count => count + 1);
                    redraw();
                  }, undefined, recover);
                };
                loadDetailMap('normal', torsoNormalReview && imported.face ? 'athlete-torso-normal-review.png'
                  : preferred && imported.face ? 'athlete-face-relief-normal.png' : 'athlete-normal.png');
                loadDetailMap('orm', 'athlete-orm.png');
                if (imported.regionalDermal) {
                  new THREE.TextureLoader().load('/assets/athlete-lip-response.png', texture => {
                    if (!isCurrent() || !installAvatarLipResponse(imported.lipSurfaces, imported.materials.lips, texture)) {
                      texture.dispose();
                      return;
                    }
                    styleImportedFigure(imported, figureSettingsRef.current);
                    element.dataset.lipResponse = 'microgrooves-v1';
                    redraw();
                  }, undefined, () => { /* Keep natural matte lips on map failure. */ });
                }
                if (preferred) {
                  new THREE.TextureLoader().load('/assets/athlete-scan-pores.png', texture => {
                    if (!isCurrent()) {
                      texture.dispose();
                      return;
                    }
                    texture.anisotropy = Math.min(4, renderer?.capabilities.getMaxAnisotropy() ?? 1);
                    installAvatarScanDetail(imported.materials.body, texture);
                    element.dataset.scanDetail = 'scan-microrelief';
                    if (import.meta.env.DEV && imported.materials.body.bumpMap?.userData.BODYFORGE_metricSkin === METRIC_SKIN_ID) {
                      element.dataset.metricSkinReview = METRIC_SKIN_ID;
                    }
                    redraw();
                  }, undefined, () => { /* Keep the procedural fallback. */ });
                }
                if (preferred) {
                  let readyFabricMaps = 0;
                  for (const [kind, file] of [['normal', 'athlete-fabric-normal.jpg'], ['roughness', 'athlete-fabric-roughness.jpg']] as const) {
                    new THREE.TextureLoader().load('/assets/' + file, texture => {
                      if (!isCurrent() || !applyAvatarFabricMap(imported.garments, kind, texture)) {
                        texture.dispose();
                        return;
                      }
                      texture.anisotropy = Math.min(4, renderer?.capabilities.getMaxAnisotropy() ?? 1);
                      texture.needsUpdate = true;
                      element.dataset.fabricMaps = String(++readyFabricMaps);
                      redraw();
                    }, undefined, () => { /* Keep the valid remaining textile maps. */ });
                  }
                }
                const brows = imported.garments.find(garment => garment.name === 'BODYFORGE_Brows');
                const facialBody = imported.surfaces.find(surface => surface.name === 'BODYFORGE_Body');
                if (preferred && imported.face && brows && facialBody) {
                  const loadBrows = (directional: boolean) => {
                    if (!isCurrent()) return;
                    const recover = () => {
                      if (directional && isCurrent()) loadBrows(false);
                    };
                    const file = directional ? 'bodyforge-brows-v3.glb' : 'bodyforge-brows-v2.glb';
                    new GLTFLoader().load('/assets/' + file, ({ scene: donor }) => {
                      if (!isCurrent() || !installAvatarBrows(facialBody, brows, donor)) {
                        disposeThreeTree(donor);
                        recover();
                        return;
                      }
                      element.dataset.browDetail = 'fitted-strands';
                      element.dataset.browRevision = String(brows.geometry.userData.BODYFORGE_brow_revision);
                      redraw();
                    }, undefined, recover);
                  };
                  loadBrows(true);
                }
                const shoes = imported.garments.find(garment => garment.name === 'BODYFORGE_Shoes');
                if (preferred && shoes) {
                  new GLTFLoader().load('/assets/bodyforge-trainers.glb', ({ scene: donor }) => {
                    if (!isCurrent() || !installAvatarTrainers(shoes, donor)) {
                      disposeThreeTree(donor);
                      return;
                    }
                    element.dataset.trainerDetail = 'fitted-geometry';
                    redraw();
                  }, undefined, () => { /* Keep the original rigged footwear. */ });
                }
                const loadFallbackEyes = () => loadDetailMap('eyes', 'athlete-eyes.jpg');
                const eyes = imported.eyes;
                if (preferred && eyes) {
                  new GLTFLoader().load('/assets/bodyforge-ocular-detail.glb', ({ scene: donor }) => {
                    ownAvatarOcularImages(donor);
                    if (!isCurrent() || !installAvatarOcularDetail(eyes, donor)) {
                      disposeThreeTree(donor);
                      loadFallbackEyes();
                      return;
                    }
                    element.dataset.ocularDetail = 'fitted-pbr';
                    setDetailMaps(count => count + 1);
                    redraw();
                  }, undefined, loadFallbackEyes);
                } else loadFallbackEyes();
              }
            }
            runtimeDiagnostics.recordGlbLoad(telemetryId, performance.now() - glbStartedAt);
            redraw();
          } catch (error) {
            console.warn('CC0 avatar unavailable, retaining procedural fallback:', error);
            disposeThreeTree(gltf.scene);
            if (fallback.length) loadFigure(fallback);
          }
        },
        undefined,
        () => { if (!disposed && fallback.length) loadFigure(fallback); },
      );
      // Compact attribute encoding keeps the 163-bone rig and all 129,973
      // triangles. Keep the original full-precision GLB as the first fallback.
      loadFigure(['/assets/bodyforge-athlete-efficient.glb', '/assets/bodyforge-athlete-detail.glb', '/assets/bodyforge-athlete-realistic.glb', '/assets/bodyforge-athlete.glb', '/assets/male-base-mesh-base_male.glb']);
    } catch (error) {
      console.error('BODYFORGE 3D renderer unavailable:', error);
      renderer?.dispose();
      portraitShadow?.dispose();
      studio?.dispose();
      runtimeDiagnostics.detachAvatar(telemetryId);
      if (!disposed) setAvailable(false);
    }

    return () => {
      runtime.current?.stopped();
      runtime.current = null;
    };
  }, []);

  useEffect(() => {
    const state = runtime.current;
    if (!state) return;
    state.setAccent(lookColors(look).trim);
    if (importedRef.current) {
      styleImportedFigure(importedRef.current, { build, muscles, outfit, look });
      state.redraw();
      return;
    }
    if (state.current) {
      state.figureRoot.remove(state.current);
      disposeThreeTree(state.current);
    }
    const mesh = makeFigure({ build, muscles, outfit, look });
    state.figureRoot.add(mesh);
    state.current = mesh;
    state.redraw();
  }, [build, muscles.arms, muscles.core, muscles.legs, muscles.back, outfit, look]);

  useEffect(() => {
    runtime.current?.setView(focus);
    runtime.current?.setYaw(yaw);
  }, [focus, yaw]);

  if (!available) {
    return (
      <div className={`avatar-3d-fallback ${className}`} data-3d-status="fallback">
        <Avatar workouts={0} build={build} muscles={muscles} outfit={outfit} look={look}/>
        <span>3D сейчас недоступно — включён облегчённый персонаж</span>
      </div>
    );
  }

  return (
    <div className={`avatar-3d-shell ${className}`} data-3d-status="active" data-avatar-source={modelSource}
      data-avatar-asset={modelAsset}
      data-skin-map={skinMapReady ? 'ready' : 'basic'}
      data-avatar-revision={groomed ? 'detail-v2' : realistic ? 'realistic-v1' : 'base'}
      data-hair-maps={hairMaps}
      data-detail-maps={detailMaps}
      data-outfit-fit={modelAsset === 'athlete' ? 'skinned-garment' : modelSource === "cc0" ? "skinned-surface" : "procedural"}
      data-look-skin={look.skin} data-look-hair={look.hair} data-look-hair-style={look.hairStyle ?? 'crop'}
      data-look-trim={look.trim} data-look-shirt={look.shirt === true ? 'on' : 'off'} data-look-visor={look.visor ? "on" : "off"}>
      <div className="avatar-3d-embers" aria-hidden="true" />
      <canvas ref={canvas} tabIndex={0} aria-keyshortcuts="ArrowLeft ArrowRight"
        aria-label="Объёмный 3D-персонаж BODYFORGE. Поверни пальцем или клавишами со стрелками влево и вправо." role="img" />
      <span className="avatar-3d-drag">↔ ПОВЕРНИ ГЕРОЯ</span>
      <span className="avatar-3d-label">BODYFORGE / 3D БОЕЦ</span>
    </div>
  );
}

/** Anatomical CC0 GLB display. It is never sampled from the phone camera.
 * Fabric is painted on the original skinned surface; hair/visor use head bone. */
function buildImportedFigure(source: THREE.Group, settings: FigureParams, preferred: boolean, scanReview?: (source: THREE.Object3D) => number): ImportedFigure {
  const root = new THREE.Group();
  const body = new THREE.Group();
  root.add(body);

  // This pinned CC0 MakeHuman GLB is authored in metres, with feet at zero.
  // Normalize the original skinned mesh ONCE. Its vertex coordinates and bone
  // weights remain untouched, so subsequent fitted colors follow every limb.
  // Hair length must not change body normalization or camera framing.
  source.updateWorldMatrix(true, true);
  const box = new THREE.Box3();
  source.traverse((node) => {
    if (node instanceof THREE.Mesh && !node.name.startsWith('BODYFORGE_Hair_')) box.expandByObject(node);
  });
  const dimensions = box.getSize(new THREE.Vector3());
  if (!Number.isFinite(dimensions.y) || dimensions.y <= .1) {
    throw new Error('CC0 character has invalid dimensions');
  }
  const scale = 4.75 / dimensions.y;
  const centre = box.getCenter(new THREE.Vector3());
  source.scale.setScalar(scale);
  source.position.set(-centre.x * scale, -box.min.y * scale + .028, -centre.z * scale);
  body.add(source);

  // The derived GLB supplies real skinned garments. The original GLB remains
  // a compatible fallback with a painted fitted layer.
  const materials = createAvatarMaterials();
  const surfaces: THREE.SkinnedMesh[] = [];
  const nailSurfaces: THREE.SkinnedMesh[] = [];
  const lipSurfaces: THREE.SkinnedMesh[] = [];
  const garments: THREE.SkinnedMesh[] = [];
  let weave: THREE.DataTexture | undefined;
  const replacedTextiles = new Set<THREE.Material>();
  const authoredHair: Partial<Record<HairStyle, THREE.SkinnedMesh>> = {};
  const eyeMeshes: THREE.SkinnedMesh[] = [];
  let realistic = false;
  let groomed = false;
  let headBone: THREE.Bone | null = null;
  source.traverse((node) => {
    if (node instanceof THREE.Bone) {
      if (node.name.toLowerCase() === 'head') headBone = node;
    }
    if (node instanceof THREE.SkinnedMesh) {
      if (!node.geometry.getAttribute('position') || !node.geometry.getAttribute('skinIndex')) {
        throw new Error('CC0 skinned mesh lacks positions or skin weights');
      }
      if (node.name === 'BODYFORGE_Body') {
        realistic = node.userData.BODYFORGE_realism === 'v1';
        groomed = node.userData.BODYFORGE_groom === 'v2';
      }
      const style = /^BODYFORGE_Hair_(crop|buzz|swept)$/.exec(node.name)?.[1] as HairStyle | undefined;
      if (style) {
        node.castShadow = true;
        authoredHair[style] = node;
        return;
      }
      const garment = /^BODYFORGE_(Shirt|Shorts|Shoes|ChestMark|Brows)$/.test(node.name);
      const role = garment ? 'garment' : importedSurfaceRole(node.name, node.geometry);
      if (garment) {
        garments.push(node);
        const material = node.material as THREE.MeshStandardMaterial;
        if (node.name !== 'BODYFORGE_ChestMark' && node.name !== 'BODYFORGE_Brows') {
          const textile = isAvatarTextile(node);
          // Clothing owns a separate weave; the skin's microrelief may be replaced asynchronously.
          weave ??= createAvatarFabric();
          if (textile && preferred && material instanceof THREE.MeshStandardMaterial) {
            node.material = createAvatarTextileMaterial(material, weave);
            replacedTextiles.add(material);
          } else {
            material.bumpMap = weave;
            material.bumpScale = textile ? .004 : .0015;
            if (textile) material.roughness = .76;
          }
        }
      } else node.material = materials[role as keyof AvatarMaterials];
      node.userData.surfaceRole = role;
      if (role === 'eyes') {
        paintAvatarEyes(node.geometry);
        eyeMeshes.push(node);
      }
      if (role === 'body') surfaces.push(node);
      if (role === 'nails') nailSurfaces.push(node);
      if (role === 'lips') lipSurfaces.push(node);
      node.castShadow = role === 'body' || /^BODYFORGE_(Shirt|Shorts|Shoes)$/.test(node.name);
      node.receiveShadow = true;
    }
  });
  // Only release replaced material instances after checking all mesh owners.
  source.traverse(node => {
    if (node instanceof THREE.Mesh) {
      for (const material of Array.isArray(node.material) ? node.material : [node.material]) replacedTextiles.delete(material);
    }
  });
  replacedTextiles.forEach(material => material.dispose());
  if (surfaces.length === 0 || !headBone) {
    Object.values(materials).forEach((material) => material.dispose());
    materials.body.bumpMap?.dispose();
    throw new Error('CC0 rig cannot provide skinned body and head anchor');
  }
  // Keep both torso states on one indexed mesh. Unknown rigs fail closed.
  const wardrobe = createAthleteWardrobe(source);
  if (!wardrobe) maskAthleteOcclusion(source);

  // The real head bone is at approximately (0,4.35,-.26) *after*
  // normalization, whereas the procedural hero's head was assumed to be at
  // (0,4.58,0). That was the actual source of floating hair and visor.
  const outfitAnchor = new THREE.Group();
  body.add(outfitAnchor);
  body.updateMatrixWorld(true);
  const headPosition = outfitAnchor.worldToLocal(
    (headBone as THREE.Bone).getWorldPosition(new THREE.Vector3()),
  );

  const hair = createAvatarHairMaterial(0x191b21, realistic);
  const buzzHair = groomed ? createAvatarHairMaterial(0x191b21, true) : hair;
  const hairStyles = {} as Record<HairStyle, THREE.Object3D>;
  const useAuthoredHair = HAIR_STYLES.every(({ id }) => authoredHair[id]);
  for (const { id } of HAIR_STYLES) {
    if (useAuthoredHair) {
      const style = authoredHair[id]!;
      style.material = id === 'buzz' ? buzzHair : hair;
      hairStyles[id] = style;
      continue;
    }
    if (authoredHair[id]) authoredHair[id]!.visible = false;
    const style = createAvatarHair(id, hair);
    // Shrink the procedural hairstyle to the measured MakeHuman skull.
    // Keep its world-space silhouette, then reparent to the actual head bone.
    style.scale.set(.72, 1.07, .74);
    const prototypeY = id === 'buzz' ? 4.620 : 4.576;
    style.position.set(
      headPosition.x,
      headPosition.y + .20 - prototypeY * 1.07,
      headPosition.z + .08,
    );
    outfitAnchor.add(style);
    (headBone as THREE.Bone).attach(style);
    hairStyles[id] = style;
  }

  // The visor is fitted over the forehead/eye line, not the model origin.
  const visorRoot = new THREE.Group();
  visorRoot.position.copy(headPosition).add(new THREE.Vector3(0, .055, .255));
  outfitAnchor.add(visorRoot);
  const visorMaterial = new THREE.MeshStandardMaterial({
    color: '#183648', metalness: .12, roughness: .24,
    transparent: true, opacity: .38, depthWrite: false,
  });
  const visor = new THREE.Mesh(new THREE.BoxGeometry(.470, .077, .025), visorMaterial);
  visor.name = 'holo-visor-glass';
  visorRoot.add(visor);
  const visorEdgeMaterial = new THREE.MeshStandardMaterial({
    color: '#55e3f2', metalness: .36, roughness: .24,
    emissive: '#55e3f2', emissiveIntensity: .3,
  });
  const visorEdge = new THREE.Mesh(new THREE.BoxGeometry(.483, .012, .035), visorEdgeMaterial);
  visorEdge.name = 'holo-visor-frame';
  visorEdge.position.y = .044;
  visorRoot.add(visorEdge);
  (headBone as THREE.Bone).attach(visorRoot);

  applyAthleticPose(source);
  source.updateWorldMatrix(true, true);
  const groomAnchor = new THREE.Group();
  groomAnchor.matrixAutoUpdate = false;
  const head = headBone as THREE.Bone;
  groomAnchor.matrix.copy(head.matrixWorld).invert().multiply(source.matrixWorld);
  head.add(groomAnchor);
  const face = preferred ? applyAvatarFaceForm(source) : undefined;
  if (face) applyAvatarFaceContour(source);
  if (face) scanReview?.(source);
  const bodyDetail = !(import.meta.env.DEV && new URLSearchParams(window.location.search).get('avatarBodyDermal') === 'base');
  const regionalDermal = Boolean(face && installDermalRegions(surfaces, bodyDetail));
  if (regionalDermal) {
    materials.body.userData.BODYFORGE_regionalDermal = true;
    if (import.meta.env.DEV &&
        (window.location.hostname === '127.0.0.1' || window.location.hostname === 'localhost') &&
        new URLSearchParams(window.location.search).get('avatarMetricSkin') === '1') {
      installMetricSkinCoordinates(surfaces, materials.body, scale);
    }
    if (import.meta.env.DEV && new URLSearchParams(window.location.search).get('avatarSkinDiffusion') === '1') {
      materials.body.userData.BODYFORGE_diffusionReview = true;
    }
    installScalpDensity(surfaces, materials.body);
    if (import.meta.env.DEV && new URLSearchParams(window.location.search).get('avatarBuzzSurface') === '1') {
      installBuzzSurface(materials.body);
    }
    installAvatarSkinLighting(materials.body);
  }
  const lashBody = surfaces.find(surface => surface.name === 'BODYFORGE_Body');
  const lashes = face && lashBody ? createAvatarLashes(lashBody) : undefined;
  const seams = preferred ? createAvatarGarmentSeams(garments) : undefined;
  const garmentFit = seams ? applyAvatarGarmentFit(garments) : 0;
  const sleeveRepair = garmentFit ? repairAvatarSleeve(garments) : 0;
  const landmarksReview = preferred && import.meta.env.DEV && new URLSearchParams(window.location.search).get('avatarLandmarks') === '1';
  const scanTorsoReview = preferred && import.meta.env.DEV && new URLSearchParams(window.location.search).get('avatarScanTorso') === '1'
    ? SCAN_TORSO_REVIEW_DATA : undefined;
  const athleticReview = import.meta.env.DEV && Boolean(scanTorsoReview) &&
    new URLSearchParams(window.location.search).get('avatarAthletic') === '1';
  const reliefReview = athleticReview && import.meta.env.DEV &&
    (window.location.hostname === '127.0.0.1' || window.location.hostname === 'localhost') &&
    new URLSearchParams(window.location.search).get('avatarMuscleRelief') === '1';
  const sculpt = groomed ? createAvatarPhysique(source, Boolean(seams), landmarksReview, scanTorsoReview, athleticReview, reliefReview) : undefined;
  if (typeof source.userData.BODYFORGE_muscleReliefReview === 'string') {
    root.userData.BODYFORGE_muscleReliefReview = source.userData.BODYFORGE_muscleReliefReview;
  }
  if (typeof source.userData.BODYFORGE_athleticReview === 'string') {
    root.userData.BODYFORGE_athleticReview = source.userData.BODYFORGE_athleticReview;
  }
  if (typeof source.userData.BODYFORGE_scanTorsoRelief === 'string') {
    root.userData.BODYFORGE_scanTorsoRelief = source.userData.BODYFORGE_scanTorsoRelief;
  }
  const idle = preferred && seams ? createAvatarIdle(source) : undefined;

  const imported: ImportedFigure = {
    root, surfaces, nailSurfaces, lipSurfaces, garments, wardrobe, materials, hair, buzzHair, hairStyles, realistic, groomed, groomAnchor, sculpt, seams, idle, idleBreath: 0,
    eyes: eyeMeshes.length === 1 ? eyeMeshes[0] : undefined,
    face, lashes, regionalDermal, garmentFit, sleeveRepair,
    visor, visorMaterial, visorEdge, visorEdgeMaterial,
  };
  styleImportedFigure(imported, settings);
  return imported;
}

function styleAvatarHair(imported: ImportedFigure, look: AvatarLook) {
  const color = lookColors(look).hair;
  const selected = look.hairStyle ?? 'crop';
  const buzzSurface = import.meta.env.DEV &&
    setBuzzSurface(imported.materials.body, selected === 'buzz' && imported.strandGroom?.directional === true);
  imported.hair.color.set(color);
  imported.buzzHair.color.set(color);
  for (const style of HAIR_STYLES) {
    imported.hairStyles[style.id].visible = style.id === selected && (!imported.strandGroom || selected === 'buzz') && !buzzSurface;
  }
  if (imported.strandGroom) {
    const { crop, swept, material } = imported.strandGroom;
    material.color.set(color).lerp(new THREE.Color('#6f6156'), .13);
    crop.visible = selected === 'crop';
    swept.visible = selected === 'swept';
    setScalpDensityLook(imported.materials.body, material.color, imported.strandGroom.layered && (selected !== 'buzz' || buzzSurface));
  }
}

function styleImportedFigure(imported: ImportedFigure, { build, muscles, outfit, look = DEFAULT_AVATAR_LOOK }: FigureParams) {
  const colors = lookColors(look);
  imported.wardrobe?.setShirt(look.shirt === true);
  // Outfit follows the *same skinned GLB positions*, not independent meshes.
  for (const surface of imported.surfaces) {
    if (imported.garments.length === 0) paintFittedBody(surface.geometry, { skin: colors.skin, trim: colors.trim, outfit, shirt: look.shirt === true });
    else {
      const geometry = surface.geometry;
      const count = geometry.getAttribute('position').count;
      let attribute = geometry.getAttribute('color');
      if (!attribute || attribute.count !== count) {
        attribute = new THREE.BufferAttribute(new Float32Array(count * 3), 3);
        geometry.setAttribute('color', attribute);
      }
      const skin = new THREE.Color(colors.skin);
      if (imported.materials.body.map) {
        const reference = new THREE.Color('#edbf9d');
        skin.r /= reference.r; skin.g /= reference.g; skin.b /= reference.b;
      }
      for (let i = 0; i < count; i++) attribute.setXYZ(i, skin.r, skin.g, skin.b);
      if (imported.face?.body === geometry) tintAvatarFace(attribute, imported.face.pigment, skin);
      attribute.needsUpdate = true;
    }
  }
  for (const surface of imported.nailSurfaces) tintAvatarNails(surface.geometry, colors.skin);
  for (const surface of imported.lipSurfaces) tintAvatarLips(surface.geometry, colors.skin, !!imported.materials.lips.map);
  for (const garment of imported.garments) {
    const material = garment.material as THREE.MeshStandardMaterial;
    const shirt = garment.name === 'BODYFORGE_Shirt';
    material.color.set(garment.name === 'BODYFORGE_Brows' ? colors.hair :
      garment.name === 'BODYFORGE_ChestMark' ? colors.trim :
      garment.name === 'BODYFORGE_Shoes' ? '#202b39' :
      outfit === 'steel' ? (shirt ? '#263e53' : '#354e61') : outfit === 'ember' ?
        (shirt ? '#453130' : '#764d39') : (shirt ? '#17212d' : '#161d27'));
    if (isAvatarTextile(garment) && material instanceof THREE.MeshPhysicalMaterial) {
      tintAvatarTextile(material);
    }
  }
  styleAvatarHair(imported, look);
  imported.visorEdgeMaterial.color.set(colors.trim);
  imported.visorEdgeMaterial.emissive.set(colors.trim);
  imported.visor.visible = look.visor;
  imported.visorEdge.visible = look.visor;
  imported.root.scale.x = build === 'solid' ? 1.055 : 1;
  imported.sculpt?.(muscles);
  applyAvatarGrowth(imported.root, muscles, imported.idleBreath);
  imported.seams?.update(imported.root);
}

function makeFigure({ build, muscles, outfit, look = DEFAULT_AVATAR_LOOK }: FigureParams): THREE.Group {
  const colors = lookColors(look);
  const base = new THREE.Color(colors.skin);
  const arms = muscles.arms ?? 0;
  const core = muscles.core ?? 0;
  const legs = muscles.legs ?? 0;
  const back = muscles.back ?? 0;
  const solid = build === 'solid' ? 1 : 0;
  const group = new THREE.Group();
  const sphere = new THREE.SphereGeometry(1, 32, 24);
  // Restrained PBR skin and anatomy. Growth still follows game.muscles;
  // unlike the previous spherical joints, the figure reads as one human.
  const skin = new THREE.MeshStandardMaterial({ color: base, roughness: .78, metalness: 0 });
  const warmSkin = new THREE.MeshStandardMaterial({ color: base.clone().lerp(new THREE.Color(0xffffff), .07), roughness: .76, metalness: 0 });
  const skinShade = new THREE.MeshStandardMaterial({ color: base.clone().lerp(new THREE.Color(0x000000), .22), roughness: .83, metalness: 0 });
  const hair = createAvatarHairMaterial(colors.hair);
  const faceDetail = new THREE.MeshStandardMaterial({ color: 0x382a28, roughness: .92 });
  const outfitColor = outfit === 'ember' ? 0x976742 : outfit === 'steel' ? 0x476275 : 0x1b2733;
  const fabric = new THREE.MeshStandardMaterial({ color: outfitColor, roughness: .71, metalness: .09 });
  const fabricShadow = new THREE.MeshStandardMaterial({ color: outfit === 'ember' ? 0x511e16 : 0x19242b, roughness: .72 });
  const trim = new THREE.MeshStandardMaterial({
    color: colors.trim, metalness: outfit === 'base' ? .45 : .82, roughness: .22,
  });
  const shoe = new THREE.MeshStandardMaterial({ color: outfit === 'steel' ? 0x43525a : 0x1f272a, roughness: .55 });
  const jersey = new THREE.MeshStandardMaterial({ color: outfit === 'ember' ? 0x423130 : outfit === 'steel' ? 0x233647 : 0x18232e,
    roughness: .9, side: THREE.DoubleSide });
  const shoeSole = new THREE.MeshStandardMaterial({ color: 0xd8c5aa, roughness: .68 });

  function ball(name: string, x: number, y: number, z: number, sx: number, sy: number, sz: number,
    material: THREE.Material): THREE.Mesh {
    const part = new THREE.Mesh(sphere, material);
    part.name = name;
    part.position.set(x, y, z);
    part.scale.set(sx, sy, sz);
    group.add(part);
    return part;
  }
  function limb(name: string, start: [number, number, number], end: [number, number, number],
    width: number, depth: number, material: THREE.Material) {
    const a = new THREE.Vector3(...start);
    const b = new THREE.Vector3(...end);
    const length = a.distanceTo(b);
    // Smooth anatomical taper: continuous limb profile, not a scaled bead.
    // The ends overlap the joints, so the silhouette stays connected.
    const rings = [
      [-.52, .68], [-.44, .83], [-.29, .97],
      [-.1, 1], [.12, .94], [.33, .85], [.52, .72],
    ];
    const geometry = new THREE.LatheGeometry(
      rings.map(([offset, radius]) => new THREE.Vector2(width * radius, length * offset)), 32,
    );
    const part = new THREE.Mesh(geometry, material);
    part.name = name;
    part.position.copy(a).add(b).multiplyScalar(.5);
    part.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), b.sub(a).normalize());
    part.scale.z = depth / width;
    group.add(part);
    return part;
  }
  const spread = solid ? .065 : 0;
  const legMass = legs*.023 + solid*.025;
  const waist = .48 + spread + core*.022;
  const shoulderRadius = .66 + back*.025 + solid*.055;

  // Legs: visible thighs, knees, sculpted calves and thick soles.
  for(const side of [-1, 1]) {
    const hip = side*(.37+spread*.3);
    const knee = side*.385;
    const ankle = side*.405;
    limb('thigh', [hip,2.25,.035],[knee,1.34,.07],.258+legMass,.245+legMass*.58,skin);
    ball('knee', knee,1.35,.085,.176+legMass*.28,.165,.175, skin);
    limb('shin',[knee,1.29,.045],[ankle,.43,.04],.162+legMass*.45,.166+legMass*.48,skin);
    ball('calf', side*.398,.95,-.085,.168+legMass*.37,.255,.16+legMass*.3,skin);
    ball('ankle', ankle,.31,.05,.135,.18,.14,skin);
    ball('shoe', ankle,.145,.23,.255,.14,.39,shoe);
    ball('sole',ankle,.073,.24,.266,.047,.398,shoeSole);
    ball('lace',ankle,.255,.385,.095,.009,.13,shoeSole);
  }

  // Hips, belt and base clothing.
  ball('pelvis',0,2.28,0,.60+spread,.44,.32,skin);
  ball('shorts',0,2.26,.025,.585+spread,.33,.34,fabric);
  ball('shorts-underside',0,2.12,-.11,.56+spread,.18,.26,fabricShadow);
  ball('belt',0,2.54,.04,.57+spread,.049,.33,fabricShadow);
  for(const side of [-1,1]) {
    ball('short-leg',side*.36,2.04,.04,.285+legMass*.22,.24,.29,fabric);
    ball('short-hem',side*.36,1.88,.04,.283+legMass*.22,.022,.29,fabricShadow);
  }

  // Torso is a continuous sculpted lathe, not a flat plane.
  const torsoRings = [
    [waist*.65,2.51],[waist*.91,2.73],[waist,2.96],
    [.535+spread,3.16],[shoulderRadius*.94,3.45],
    [shoulderRadius,3.71],[.405+back*.012,3.91],
    [.205,4.015]
  ] as const;
  const lathe = new THREE.LatheGeometry(
    torsoRings.map(([r,y])=>new THREE.Vector2(r,y)),32,
  );
  const torso = new THREE.Mesh(lathe,skin);
  torso.name='muscular-torso';
  torso.scale.z = .67;
  group.add(torso);
  ball('abdomen',0,2.97,.16+solid*.03,waist*.89,.37,.125+solid*.03,skin);
  // The same universal default workout jersey on procedural devices.
  const shirtGeometry = new THREE.LatheGeometry([
    new THREE.Vector2(waist*.85,2.56),new THREE.Vector2(waist+ .055,2.71),
    new THREE.Vector2(waist+ .075,3.08),new THREE.Vector2(.59+spread,3.35),
    new THREE.Vector2(shoulderRadius+.035,3.66),new THREE.Vector2(.45,3.86),
    new THREE.Vector2(.25,3.99),
  ],32);
  const jerseyBody = new THREE.Mesh(shirtGeometry,jersey);
  jerseyBody.name='technical-jersey';
  jerseyBody.scale.z=.79;
  jerseyBody.visible=look.shirt === true;
  group.add(jerseyBody);
  ball('shirt-shoulder-yoke',0,3.81,.025,.69,.185,.31,jersey).visible=look.shirt === true;
  const chestMass=.022+back*.015;
  for(const side of [-1,1]){
    ball('pectoral',side*(.27+back*.008),3.52,.255,.282+chestMass,.145+back*.011,.119+chestMass*.4,skin);
    ball('collar',side*.23,3.88,.11,.25,.065,.20,skin);
  }
  // Rising muscle ridges. Higher core level changes real mesh depth.
  for(let row=0;row<3;row++){
    for(const side of [-1,1]){
      const b=ball('abdominal-ridge',side*.15,3.20-row*.20,.34+core*.005,
        .105+core*.006,.059+core*.005,.023+core*.012,
        core>=1 ? warmSkin : skin);
      b.visible=core>=1;
    }
  }
  for(const side of [-1,1]){
    ball('oblique',side*(waist-.09),2.99,.20,.115+core*.012,.24,.105,skin);
  }
  // Shoulders and arms are individually reshaped with the "arms" stat.
  for(const side of [-1,1]){
    const sx=side*(shoulderRadius+.08);
    ball('shirt-sleeve',sx,3.66,.0,.205,.18,.232,jersey).visible=look.shirt === true;
    ball('deltoid',sx,3.66,.0,.228+arms*.018,.237+arms*.017,.23+arms*.017,skin);
    limb('upper-arm',[sx,3.53,.005],[side*(.86+back*.012),2.88,.075],
      .174+arms*.020,.167+arms*.018,skin);
    ball('biceps',side*(.855+back*.012),3.19,.115,.153+arms*.02,.22,.139+arms*.015,skin);
    ball('elbow',side*(.86+back*.012),2.82,.075,.127+arms*.012,.13,.127,skin);
    limb('forearm',[side*(.86+back*.012),2.79,.07],[side*.88,2.15,.145],
      .131+arms*.013,.132+arms*.010,skin);
    ball('hand',side*.88,2.02,.155,.121,.176,.105,skin);
    if (outfit !== 'base') ball('wrist-guard',side*.88,2.30,.13,.15,.053,.155,trim);
  }

  // Neck and sculpted head: ears, jaw, facial details and hair.
  ball('neck',0,4.055,.02,.155,.215,.15,skinShade);
  ball('head',0,4.49,.07,.299,.381,.286,warmSkin);
  ball('jaw',0,4.27,.155,.205,.141,.202,skin);
  for(const side of [-1,1]){
    ball('ear',side*.298,4.46,.065,.052,.107,.057,skin);
    ball('eyebrow',side*.129,4.55,.345,.093,.017,.019,hair);
    ball('eye',side*.129,4.502,.357,.036,.019,.014,faceDetail);
  }
  ball('nose',0,4.395,.352,.040,.086,.058,skin);
  ball('mouth',0,4.265,.36,.072,.010,.014,skinShade);
  group.add(createAvatarHair(look.hairStyle ?? 'crop', hair, .064));
  if (look.visor) {
    const tinted = new THREE.MeshStandardMaterial({ color: '#203749', metalness: .12, roughness: .25,
      transparent: true, opacity: .42, depthWrite: false });
    const visor = new THREE.Mesh(new THREE.BoxGeometry(.61, .085, .04), tinted);
    visor.name = 'holographic-visor';
    visor.position.set(0, 4.52, .366);
    group.add(visor);
    const rim = new THREE.Mesh(new THREE.BoxGeometry(.62, .013, .05),
      new THREE.MeshStandardMaterial({ color: colors.trim, emissive: colors.trim, emissiveIntensity: .27 }));
    rim.name = 'visor-edge';
    rim.position.set(0, 4.57, .385);
    group.add(rim);
  }

  // Outfits have visible 3D details instead of a texture-only color change.
  if(outfit!=='base'){
    for(const side of [-1,1]){
      ball('shoulder-brand',side*(shoulderRadius+.08),3.79,.18,.084,.026,.065,trim).visible=look.shirt === true;
      ball('shin-brace',side*.405,.41,.16,.155,.064,.14,fabric);
      ball('shoe-plate',side*.405,.225,.52,.12,.025,.075,trim);
    }
  }
  // A chest emblem appears after a couple of upgrades.
  if(arms+back+core+legs>=8) {
    ball('forged-emblem',0,3.67,.40,.085,.085,.023,trim).visible=look.shirt === true;
  }
  return group;
}
